YouTube MP4 Converter (PyScript + FFmpeg WASM)

※ 경고: 이 코드는 YouTube의 서비스 약관 및 저작권 정책을 위반할 수 있으므로, 학습 및 실험 목적으로만 사용하십시오.


from pyodide import to_js from pyodide.http import pyfetch import js import re # CORS 프록시 (필요시 변경; 또는 직접 CORS 허용 서버 운영) cors_proxy = "https://thingproxy.freeboard.io/fetch/" async def fetch_html(url: str) -> str: """ 주어진 URL에 대해 GET 요청을 수행하고, text(HTML)을 반환. pyfetch 사용. """ response = await pyfetch(url) if response.ok: text = await response.text() return text else: raise RuntimeError(f"HTTP Error: {response.status} {response.status_text}") async def fetch_binary(url: str) -> bytes: """ 주어진 URL에 대해 GET 요청을 수행하고, 바이너리를 반환. """ response = await pyfetch(url) if response.ok: data = await response.bytes() return data else: raise RuntimeError(f"HTTP Error: {response.status} {response.status_text}") async def pyConvert(): progressDiv = js.document.getElementById("progress") progressDiv.textContent = "프로세스 시작...\n" videoUrl = js.document.getElementById("videoUrl").value.strip() if not videoUrl: progressDiv.textContent += "유효한 URL을 입력하세요.\n" return # 1) YouTube URL에서 v= 추출 match = re.search(r"[?&]v=([^&]+)", videoUrl) if not match: progressDiv.textContent += "유효한 YouTube URL이 아닙니다.\n" return videoId = match.group(1) progressDiv.textContent += f"비디오 ID 추출됨: {videoId}\n" # 2) yt-download.org의 API 페이지 HTML 가져오기 apiUrl = f"https://www.yt-download.org/api/button/mp4/{videoId}" progressDiv.textContent += f"다운로드 페이지 요청 중...\n{apiUrl}\n" try: htmlText = await fetch_html(cors_proxy + apiUrl) except Exception as e: progressDiv.textContent += f"페이지 요청 오류: {str(e)}\n" return # 3) HTML에서 다운로드 링크 추출 # 간단히 href에 'download' 문자열이 들어간 태그를 찾음 # 정규표현식 또는 간단히 split, or idx 탐색도 가능 match_dl = re.search(r'href="([^"]*download[^"]*)"', htmlText) if not match_dl: progressDiv.textContent += "다운로드 링크를 찾을 수 없습니다.\n" return downloadLink = match_dl.group(1) progressDiv.textContent += f"다운로드 링크 발견:\n{downloadLink}\n" # 4) 영상 바이너리 다운로드 progressDiv.textContent += "비디오 파일 다운로드 중...\n" try: videoData = await fetch_binary(cors_proxy + downloadLink) size = len(videoData) progressDiv.textContent += f"비디오 파일 다운로드 완료 (크기: {size} 바이트)\n" except Exception as e: progressDiv.textContent += f"비디오 다운로드 오류: {str(e)}\n" return # 5) JS의 convertToMp4 함수에 Uint8Array로 전달 # (PyScript에서 bytes → JS Uint8Array 변환) array = js.Uint8Array.new(to_js(videoData)) progressDiv.textContent += "JS로 변환 요청...\n" js.convertToMp4(array) # 이후 진행은 JS 쪽에서 처리 (ffmpeg.wasm) # PyScript 영역에서 함수/이벤트 연결 # 자바스크립트로 document.getElementById("convertBtn").onclick = pyConvert도 가능하지만, # 아래와 같이 Python 측에서 addEventListener를 달 수도 있음. js.document.getElementById("convertBtn").addEventListener("click", pyConvert)

* 이 파일은 로컬 파일이 아닌 http:// 환경에서 열어야 합니다.
